## 一、模块定位与目标

### 1.1 业务定位

- **高智能架构**: 全Family Agent协同本身就是智能化的标杆
- **创新驱动**: 创想灵韵(创意官)推动持续的技术创新
- **认知信任为极致**: 让AI成为值得信赖的智能伙伴

### 2.2 场景二：知识管理与智能问答

**场景描述**: 构建企业知识图谱和智能问答系统，让知识触手可及。

**知识管理Agent（语枢万物+千里伯乐联合）**:

```python
class EnterpriseKnowledgeSystem:
    """
    企业知识管理系统 - RAG增强生成架构
    
    Agent协同:
    - 语枢万物: 知识抽取/图谱构建/语义理解
    - 千里伯乐: 知识推荐/个性化服务/知识传播
    - 创想灵韵: 知识创作/内容润色/形式创新
    """
    
    def __init__(self):
        self.agents = {
            'thinker': '语枢万物-思考者',
            'recommender': '千里伯乐-推荐官',
            'creator': '创想灵韵-创意官'
        }
        
        # RAG配置
        self.rag_config = {
            'embedding_model': 'text-embedding-3-large',
            'vector_db': 'Pinecone/Milvus',
            'llm_model': 'gpt-4-turbo',
            'chunk_size': 1000,
            'chunk_overlap': 200,
            'top_k_retrieval': 5,
            'similarity_threshold': 0.75
        }
        
        # 知识分类
        self.knowledge_categories = [
            '制度规范', '操作手册', '最佳实践', 
            '案例分析', '培训资料', 'FAQ'
        ]
    
    def intelligent_qa(self, question: str, user_context: Dict = None) -> Dict:
        """
        智能问答
        
        Args:
            question: 用户问题
            user_context: 用户上下文(部门/角色/历史等)
            
        Returns:
            {
                'answer': 回答内容,
                'sources': 引用来源,
                'confidence': 置信度,
                'related_questions': 相关问题推荐,
                'follow_up_suggestions': 后续建议
            }
        """
        # Step 1: 问题理解 (语枢万物)
        parsed_question = self.thinker.parse_question(question)
        intent = parsed_question['intent']
        entities = parsed_question['entities']
        
        # Step 2: 知识检索 (语枢万物+向量数据库)
        retrieved_docs = self.thinker.retrieve_knowledge(
            query=parsed_question['normalized_query'],
            top_k=self.rag_config['top_k_retrieval'],
            filters=user_context.get('department_filters') if user_context else None
        )
        
        # Step 3: 答案生成 (LLM + RAG)
        answer = self._generate_rag_answer(
            question=question,
            context=retrieved_docs,
            user_context=user_context
        )
        
        # Step 4: 相关推荐 (千里伯乐)
        related_questions = self.recommender.suggest_related_questions(
            current_question=question,
            user_history=user_context.get('query_history') if user_context else []
        )
        
        # Step 5: 后续建议 (千里伯乐)
        follow_up = self.recommender.suggest_follow_up(
            answer=answer,
            user_role=user_context.get('role') if user_context else None
        )
        
        return {
            'question_id': generate_id(),
            'question': question,
            'parsed_intent': intent,
            'answer': answer['content'],
            'sources': answer['source_documents'],
            'confidence': answer['confidence'],
            'related_questions': related_questions[:5],
            'follow_up_suggestions': follow_up[:3],
            'agents_used': [self.thinker, self.recommender]
        }

# 使用示例
knowledge_sys = EnterpriseKnowledgeSystem()

result = knowledge_sys.intelligent_qa(
    question="公司差旅报销的标准是什么？",
    user_context={
        'department': '销售部',
        'role': '销售经理',
        'query_history': ['报销流程', '发票要求']
    }
)

print(f"问题: {result['question']}")
print(f"\n回答:\n{result['answer']}")
print(f"\n引用来源: {len(result['sources'])}篇文档")
print(f"置信度: {result['confidence']*100:.1f}%")
print(f"\n相关问题:")
for q in result['related_questions']:
    print(f"  • {q}")
```
