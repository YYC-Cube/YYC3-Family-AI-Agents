### 2.2 场景二：数据资产目录与血缘分析

**场景描述**: 构建企业级数据资产目录，追踪数据血缘关系。

**数据资产管理Agent（千里伯乐主导）**:

```python
class DataAssetManagementSystem:
    """
    数据资产管理系统 - AI FAmily多Agent协同
    
    核心能力:
    1. 自动发现与注册数据资产
    2. 数据血缘关系构建与分析
    3. 数据资产价值评估
    4. 数据使用情况监控
    """
    
    def __init__(self):
        self.agents = {
            'recommender': '千里伯乐-推荐官',
            'thinker': '语枢万物-思考者',
            'navigator': '言启千行-导航员',
            'guardian': '智云守护-安全官'
        }
        
        # 数据资产分类体系
        self.asset_taxonomy = {
            'domain': ['客户', '产品', '订单', '财务', '人力', '运营'],
            'sensitivity': ['公开', '内部', '机密', '绝密'],
            'criticality': ['核心', '重要', '一般', '低'],
            'lifecycle': ['采集', '加工', '存储', '应用', '归档']
        }
    
    def build_data_lineage(self, table_name: str, depth: int = 3) -> Dict:
        """
        构建数据血缘图
        
        Args:
            table_name: 目标表名
            depth: 追溯深度
            
        Returns:
            {
                'lineage_graph': 血缘关系图,
                'upstream': 上游依赖,
                'downstream': 下游影响,
                'impact_analysis': 变更影响评估
            }
        """
        # Step 1: 直接依赖识别 (语枢万物)
        direct_dependencies = self.thinker.extract_direct_dependencies(table_name)
        
        # Step 2: 递归追溯上游 (深度优先)
        upstream = {}
        if depth > 0:
            for dep in direct_dependencies['upstream']:
                upstream[dep] = self.build_data_lineage(dep, depth-1)
        
        # Step 3: 递归追溯下游 (广度优先)
        downstream = {}
        if depth > 0:
            for consumer in direct_dependencies['downstream']:
                downstream[consumer] = self._get_downstream_consumers(consumer)
        
        # Step 4: 影响分析 (智云守护+语枢万物联合)
        impact_analysis = self._assess_change_impact(
            target=table_name,
            upstream=upstream,
            downstream=downstream
        )
        
        return {
            'target_table': table_name,
            'analysis_depth': depth,
            'direct_dependencies': direct_dependencies,
            'upstream_lineage': upstream,
            'downstream_impact': downstream,
            'impact_analysis': impact_analysis,
            'generated_at': datetime.now()
        }
    
    def _assess_change_impact(self, target, upstream, downstream):
        """变更影响评估"""
        # 影响范围统计
        affected_tables_count = len(downstream)
        affected_reports_count = self._count_affected_reports(downstream)
        affected_dashboards_count = self._count_affected_dashboards(downstream)
        
        # 风险等级判定
        risk_factors = {
            'scope': min(affected_tables_count / 10, 1.0),  # 影响面
            'criticality': self._get_table_criticality(target),  # 重要度
            'sensitivity': self._get_table_sensitivity(target),  # 敏感度
            'consumer_count': len(downstream)  # 消费者数量
        }
        
        composite_risk = (
            risk_factors['scope'] * 0.3 +
            risk_factors['criticality'] * 0.25 +
            risk_factors['sensitivity'] * 0.25 +
            min(risk_factors['consumer_count'] / 20, 1.0) * 0.2
        )
        
        if composite_risk < 0.3:
            risk_level = 'low'
        elif composite_risk < 0.6:
            risk_level = 'medium'
        elif composite_risk < 0.8:
            risk_level = 'high'
        else:
            risk_level = 'critical'
            
        return {
            'risk_level': risk_level,
            'composite_score': composite_risk,
            'affected_assets': {
                'tables': affected_tables_count,
                'reports': affected_reports_count,
                'dashboards': affected_dashboards_count
            },
            'recommended_actions': self._generate_recommendations(risk_level),
            'stakeholders_to_notify': self._identify_stakeholders(downstream)
        }

# 使用示例
dam_system = DataAssetManagementSystem()

lineage = dam_system.build_data_lineage(
    table_name="dw.fact_order_detail",
    depth=3
)

print(f"数据血缘分析结果:")
print(f"  目标表: {lineage['target_table']}")
print(f"  影响风险等级: {lineage['impact_analysis']['risk_level']}")
print(f"  影响报表数: {lineage['impact_analysis']['affected_assets']['reports']}")
```
